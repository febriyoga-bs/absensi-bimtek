var D = JSON.parse(document.getElementById('data').textContent);

/* Kode status pada kolom ke-9 sekolah dan ke-4 binaan luar K3. */
var BELUM = -1, TERIMA = 0, TOLAK = 1, PROSES = 2;
var STAT = {};

STAT[TERIMA] = {t:'Diterima',        k:'p-terima', c:'var(--s-terima)'};
/* SIAP bukan status dari berkas, melainkan turunan: sekolah yang belum
   mengirim TAPI capaian fisiknya sudah menembus 50 persen, jadi sebenarnya

var SIAP = -9;

var URUT = [SIAP, BELUM, PROSES, TOLAK, TERIMA];
function statSek(x){ return x[8] === BELUM ? (x[10] != null ? SIAP : BELUM) : x[8]; }

/* Kehadiran sekolah di lokasi Bimtek, kolom ke-16. Kode sama dengan dasbor
   LO ABT2 Tahap 3. Seluruh bagian ini hanya hidup bila dataset membawa
   D.hadirStamp; tanpa itu kartunya kembali ke peringkat capaian. */
var HADIR = {
  '3':  {t:'Sudah tiba',       k:'h-tiba'},
  '2':  {t:'Sedang antri',     k:'h-antri'},
  '1':  {t:'Bersedia hadir',   k:'h-sedia'},
  '0':  {t:'Belum konfirmasi', k:'h-belum'},
  '-3': {t:'Berhalangan',      k:'h-halangan'},
  '-1': {t:'Tidak terdata',    k:'h-kosong'}
};
var HADIR_URUT = [3, 2, 1, 0, -3, -1];
function adaHadir(){ return !!D.hadirStamp; }
/* "Sudah di lokasi" mencakup yang sudah tiba dan yang sedang antri. */
function diLokasi(x){ return x.length > 15 && (x[15] === 3 || x[15] === 2); }
function pilHadir(x){
  if(!adaHadir() || x.length < 16) return '';
  var h = HADIR[String(x[15])];
  return h ? '<span class="pil '+h.k+'">'+h.t+'</span>' : '';
}

var pelan = window.matchMedia && window.matchMedia('(prefers-reduced-motion:reduce)').matches;

var MODE = 'lo', AKTIF = -1, HOTEL_AKTIF = -1;
var LB_MODE = 'delta', AMBANG = D.ambang, FASIL_BUKA = null, DET_STAT = null;

function $(id){ return document.getElementById(id); }
function esc(t){ return String(t==null?'':t).replace(/[&<>"]/g,function(c){
  return {'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;'}[c]; }); }
function nf(n){ return String(n).replace(/\B(?=(\d{3})+(?!\d))/g,'.'); }
function low(t){ return String(t==null?'':t).toLowerCase(); }
function prov(i){ return String(P[i]||'').replace(/^Prov\.\s*/,''); }
function pct(a,b){ return b ? Math.round(a*1000/b)/10 : 0; }
function satu(n){ return n==null?'–':(Math.round(n*10)/10).toString().replace('.',','); }

function animasi(induk){
  var isi = induk.querySelectorAll('[data-w]');
  var jalan = function(){ Array.prototype.forEach.call(isi, function(el){

var fasPerLO = D.lo.map(function(){ return []; });
D.fasil.forEach(function(f,i){ f.ix = i; fasPerLO[f[1]].push(f); });

D.luar.forEach(function(x){ if(x[3]===PROSES) antrePerFasil[x[2]].push({u:x[4], nm:x[1], np:x[0], kab:x[5], k3:0}); });
/* sekolah K3 per fasil, dihitung sekali supaya halaman fasil tidak menyapu

/* ---------- lingkup ---------- */
function loDalamLingkup(){
  if(MODE === 'lo') return AKTIF >= 0 ? [AKTIF] : [];
  return D.lo.map(function(r,i){ return i; }).filter(function(i){
    return MODE === 'semua' || D.lo[i][1] === HOTEL_AKTIF; });
}
function lingkupSek(){
  if(MODE === 'lo') return AKTIF>=0 ? sekPerLO[AKTIF] : [];
  var keluar = [];
  loDalamLingkup().forEach(function(i){ keluar = keluar.concat(sekPerLO[i]); });
  return keluar;
}
function lingkupFas(){
  if(MODE === 'lo') return AKTIF>=0 ? fasPerLO[AKTIF] : [];
  var keluar = [];
  loDalamLingkup().forEach(function(i){ keluar = keluar.concat(fasPerLO[i]); });
  return keluar;
}
function info(){
  if(MODE === 'lo'){
    var r = D.lo[AKTIF];
    return { judul:r[0], atas:H[r[1]], milik:true, jumLO:1,
      peran:(r[2] ? 'Captain' : 'LO Fasil') + ' · ' + (r[6].join(', ') || 'ruangan belum tercatat'),
      sekolah:r[3], fasil:r[4], koor:r[5] };
  }
  var ls = loDalamLingkup(), fas = lingkupFas();
  var koor = {}; fas.forEach(function(f){ koor[f[3]] = 1; });
  return {
    judul: MODE==='semua' ? 'Semua LO' : 'Semua LO di hotel ini',
    atas: MODE==='semua' ? (H.length + ' hotel') : H[HOTEL_AKTIF],
    peran: ls.length + ' LO · ' + (MODE==='semua' ? (H.length + ' hotel') : H[HOTEL_AKTIF]),
    sekolah: lingkupSek().length, fasil: fas.length, koor: Object.keys(koor).length,
    jumLO: ls.length, milik:false };
}

/* ---------- hitungan inti ---------- */
/* Sebaran status Laporan 50% pada sekumpulan sekolah. */
function sebaran(sek){
  var c = {}; URUT.forEach(function(k){ c[k] = 0; });
  sek.forEach(function(x){ c[statSek(x)]++; });
  return c;
}
/* Antrean reviu milik sekumpulan fasil, termasuk binaan di luar K3.
   Alasannya: tunggakan fasil adalah seluruh laporan yang menunggu dia,
   bukan hanya yang kebetulan masuk sasaran K3. */
function antrean(fas){
  var out = [];
  fas.forEach(function(f){ out = out.concat(antrePerFasil[f.ix]); });
  return out;
}
function lewatAmbang(q){ return q.filter(function(a){ return a.u != null && a.u > AMBANG; }); }

/* Jumlah fasil bermasalah, yaitu yang punya minimal satu laporan
   menunggu reviu melewati ambang. */
function fasilBermasalah(fas){
  return fas.filter(function(f){
    return antrePerFasil[f.ix].some(function(a){ return a.u != null && a.u > AMBANG; });
  });
}

    return {i:i, nama:h, lo:D.lo.filter(function(r){ return r[1]===i; }).length};
  }).filter(function(x){ return x.lo>0; }).sort(function(a,b){ return a.nama<b.nama?-1:1; });

  $('pilihHotel').innerHTML = '<option value="-1">— Pilih hotel —</option>' +
    '<option value="-2">Semua hotel (' + D.lo.length + ' LO)</option>' +
    agg.map(function(x){ return '<option value="'+x.i+'">'+esc(x.nama)+' ('+x.lo+' LO)</option>'; }).join('');

function isiNama(){

  if(isNaN(h) || h === -1){
    sel.innerHTML = '<option value="-1">— Pilih hotel dulu —</option>';
    sel.disabled = true; $('btnBuka').disabled = true;
    $('pilihKet').textContent = 'Pilih hotel tempat Anda bertugas, lalu nama Anda.';
    return;
  }
  var semua = h === -2;
  var daftar = D.lo.map(function(r,i){ return {r:r,i:i}; })
    .filter(function(x){ return semua || x.r[1]===h; })
    .sort(function(a,b){ return low(a.r[0])<low(b.r[0])?-1:1; });
  sel.innerHTML = '<option value="-1">— Pilih nama LO —</option>' +
    '<option value="-2">Semua LO' + (semua?'':' di hotel ini') + '</option>' +
    daftar.map(function(x){
      return '<option value="'+x.i+'">'+esc(x.r[0])+(x.r[2]?' (Captain)':'') +
        (semua ? ' — ' + esc(H[x.r[1]]) : '') + '</option>'; }).join('');
  sel.disabled = false; $('btnBuka').disabled = true;
  $('pilihKet').textContent = semua
    ? (D.lo.length + ' LO di ' + H.length + ' hotel. Pilih nama Anda, atau "Semua LO" untuk melihat keseluruhan.')

function gambarBeranda(){
  var g = info(), sek = lingkupSek(), fas = lingkupFas();
  var c = sebaran(sek), q = antrean(fas), lewat = lewatAmbang(q), bad = fasilBermasalah(fas);

  $('bHotel').textContent = g.atas;
  $('bNama').textContent = g.judul;
  $('bPeran').textContent = g.peran;

  /* Angka utama adalah yang benar-benar bisa ditindaklanjuti hari ini.
     Sekolah yang fisiknya belum 50% tidak bisa mengajukan apa pun, jadi
     menagihnya hanya membuang waktu LO. */
  $('bBelum').textContent = nf(c[SIAP]);
  $('bBelumKet').textContent = 'dari ' + nf(sek.length) + ' sekolah binaan \u00b7 ' +
    nf(c[BELUM]) + ' lagi belum sampai 50% fisik';

  $('bTunggak').textContent = nf(q.length);
  $('bTunggakLbl').textContent = 'laporan menunggu reviu fasil';
  /* Lingkupnya kini sama dengan Alur Laporan 50% di Statistik: hanya sekolah
     sasaran K3. Binaan di luar K3 sengaja tidak ditampilkan di aplikasi ini. */
  $('bTunggakKet').textContent = lewat.length
    ? (nf(lewat.length) + ' sudah lewat ' + AMBANG + ' hari')
    : 'belum ada yang lewat ' + AMBANG + ' hari';

  $('kSudah').textContent = nf(sek.length - c[BELUM]);
  $('kSudahLbl').textContent = 'sudah kirim (' + satu(pct(sek.length-c[BELUM], sek.length)) + '%)';
  $('kFasil').textContent = nf(bad.length) + ' / ' + nf(fas.length);
  $('kFasilLbl').textContent = 'fasil bermasalah';

  /* Kartu ketiga: kehadiran sekolah binaan di lokasi Bimtek. Selama rekap
     kehadiran belum ada, kartunya kembali ke peringkat capaian. */
  if(adaHadir()){
    var tiba = sek.filter(diLokasi).length;
    var sedia = sek.filter(function(x){ return x.length>15 && x[15]===1; }).length;
    $('kRank').textContent = nf(tiba) + ' / ' + nf(sek.length);
    /* Sebelum ada yang tiba, label membawa angka konfirmasi supaya kartunya
       tetap berguna; begitu kedatangan mulai terisi, persennya yang tampil. */
    $('kRankLbl').textContent = tiba
      ? 'sekolah hadir (' + satu(pct(tiba, sek.length)) + '%)'
      : 'sekolah hadir · ' + nf(sedia) + ' bersedia';
  } else if(g.milik){
    var daftar = papanCapaian(), rank = 0;
    daftar.forEach(function(x,n){ if(x.i===AKTIF) rank = n+1; });
    $('kRank').textContent = rank + ' / ' + D.lo.length;
    $('kRankLbl').textContent = 'peringkat capaian';
  } else {
    $('kRank').textContent = g.jumLO;
    $('kRankLbl').textContent = 'LO tercakup';
  }

  $('bKaki').innerHTML = kakiData();
  gambarPapan();
}

function kakiData(){

  return '<span class="titik"></span>' + (ada

/* ---------- papan peringkat ---------- */
/* Capaian: persen sekolah binaan yang sudah mengirim Laporan 50%. */
/* LO yang masuk papan peringkat: hanya yang sehotel dengan lingkup terpilih,
   kecuali saat lingkupnya semua hotel. Membandingkan LO lintas hotel
   menyesatkan karena beban sekolah dan tahap tiap hotel berbeda. */
function loPapan(){
  var semua = D.lo.map(function(r,i){ return i; });
  if(MODE === 'semua') return semua;
  var ht = MODE === 'lo' ? (AKTIF >= 0 ? D.lo[AKTIF][1] : -1) : HOTEL_AKTIF;
  return semua.filter(function(i){ return D.lo[i][1] === ht; });
}
function papanCapaian(){
  return loPapan().map(function(i){
    var r = D.lo[i], sek = sekPerLO[i];
    var sudah = sek.filter(function(x){ return x[8]!==BELUM; }).length;
    return {i:i, nama:r[0], hotel:H[r[1]], n:sek.length, sudah:sudah, persen:pct(sudah, sek.length)};
  }).sort(function(a,b){
    if(b.persen!==a.persen) return b.persen-a.persen;
    if(b.sudah!==a.sudah) return b.sudah-a.sudah;
    return low(a.nama)<low(b.nama)?-1:1;
  });
}
/* Pergerakan: sekolah yang mengirim pada atau setelah tanggal baseline.
   Tanggal pengajuan ada di tiap baris sekolah, jadi keadaan pada tanggal
   mana pun bisa disusun ulang tanpa perlu menyimpan agregat baseline. */
function papanDelta(){
  var batas = D.baseline;
  return loPapan().map(function(i){
    var r = D.lo[i], sek = sekPerLO[i];
    var baru = sek.filter(function(x){ return x[11] && x[11] >= batas; }).length;
    var sudah = sek.filter(function(x){ return x[8]!==BELUM; }).length;
    return {i:i, nama:r[0], hotel:H[r[1]], n:sek.length, baru:baru, sudah:sudah,
      persen:pct(sudah, sek.length)};
  }).sort(function(a,b){
    if(b.baru!==a.baru) return b.baru-a.baru;
    if(b.persen!==a.persen) return b.persen-a.persen;
    return low(a.nama)<low(b.nama)?-1:1;
  });
}
function setMode(m){
  LB_MODE = m;
  Array.prototype.forEach.call(document.querySelectorAll('#lbMode button'), function(o){
    o.setAttribute('aria-pressed', o.getAttribute('data-mode')===m ? 'true':'false'); });
}
function gambarPapan(){
  var delta = LB_MODE === 'delta';
  var daftar = delta ? papanDelta() : papanCapaian();
  var rank = 0, saya = null;
  daftar.forEach(function(x,n){ if(x.i===AKTIF){ rank = n+1; saya = x; } });
  var maks = delta ? Math.max(1, daftar[0].baru) : 100;

  function baris(x, n, ini){
    /* Nilai nol digambar tanpa isi sama sekali; sisa bar 1% pada 97 baris
       hanya jadi derau saat papan belum bergerak. */
    var nilai = delta ? x.baru : x.persen;
    var w = nilai > 0 ? Math.max(1, Math.round(delta ? nilai*100/maks : nilai)) : 0;
    var kanan = delta ? ((x.baru>0?'+':'') + nf(x.baru)) : (satu(x.persen) + '%');
    /* Judul bagian sudah menyebut titik mulainya, jadi baris tidak perlu
       mengulangnya; cukup "belum bergerak" saat nol. */
    var kecil = delta
      ? ((x.baru>0 ? x.baru + ' sekolah kirim sejak Bimtek mulai' : 'belum bergerak') +
         ' · total ' + x.sudah + '/' + x.n + ' · ' + esc(x.hotel))
      : (x.sudah + ' dari ' + x.n + ' sekolah sudah kirim · ' + esc(x.hotel));
    return '<div class="row'+(ini?' milik':'')+'">' +
      '<div class="top"><span class="grow"><b>'+(n+1)+'. '+esc(x.nama)+'</b></span>' +
      (ini?'<span class="tag">Anda</span>':'') +
      '<b class="num '+(delta?(x.baru>0?'naik':'diam'):'')+'">'+kanan+'</b></div>' +
      '<div class="bar"><i data-w="'+w+'"></i></div>' +
      '<small>'+kecil+'</small></div>';
  }

  $('lbSaya').innerHTML = saya ? baris(saya, rank-1, true) : '';
  $('lbSaya').classList.toggle('hide', !saya);
  animasi($('lbSaya'));
  /* Pencarian hanya menyaring tampilan, bukan peringkatnya: nomor urut tiap
     baris tetap posisi sebenarnya di papan, supaya hasil cari tidak terbaca
     sebagai peringkat baru. */
  var cari = low($('lbCari').value).trim();
  var tampil = daftar.map(function(x,n){ return {x:x, n:n}; });
  if(cari) tampil = tampil.filter(function(o){
    return (low(o.x.nama) + ' ' + low(o.x.hotel)).indexOf(cari) !== -1; });
  $('lbList').innerHTML = tampil.length
    ? tampil.map(function(o){ return baris(o.x, o.n, o.x.i===AKTIF); }).join('')
    : '<div class="kosong">Tidak ada LO yang cocok dengan pencarian.</div>';
  animasi($('lbList'));
  var lingkupPapan = MODE === 'semua' ? (H.length + ' hotel')
    : H[MODE === 'lo' ? D.lo[AKTIF][1] : HOTEL_AKTIF];
  $('lbKet').textContent = daftar.length + ' LO · ' + lingkupPapan;

  var totBaru = daftar.reduce(function(s,x){ return s+(x.baru||0); },0);
  var diam = daftar.filter(function(x){ return !x.baru; }).length;
  var atas = daftar[0], bawah = daftar[daftar.length-1];
  $('lbNote').textContent = delta
    ? ('Diukur dari jumlah sekolah binaan yang mengirim Laporan 50% pada atau setelah ' +
       D.baseLabel + '. Tanggal pengajuan dibaca dari Monitoring Laporan Tengah, jadi ' +
       'angka ini ikut benar sendiri setiap kali data ditarik ulang. ' +
       (totBaru === 0
         ? 'Belum ada satu pun sekolah yang mengirim sejak titik itu, jadi papan ini masih rata ' +
           'nol dan urutannya untuk sementara mengikuti capaian. Angkanya mulai bergerak begitu ' +
           'peserta mengirim laporan dan data ditarik ulang.'
         : ((saya ? saya.nama + ' ada di peringkat ' + rank + ' dari ' + daftar.length + ' LO. ' : '') +
            'Seluruh K3 bertambah ' + nf(totBaru) + ' sekolah' +
            (diam ? ', ' + diam + ' LO belum bergerak' : '') + '.')))
    : ('Persen sekolah binaan yang sudah mengirim Laporan 50%, apa pun status reviunya. ' +
       (saya ? saya.nama + ' ada di peringkat ' + rank + ' dari ' + daftar.length + ' LO dengan ' +

function gambarWilayah(){
  var sek = sekLingkupPilihan().sek, kab = {};
  sek.forEach(function(x){
    if(!kab[x[3]]) kab[x[3]] = {nama:x[3], n:0, c:{}};
    var v = kab[x[3]]; v.n++;
    v.c[x[8]] = (v.c[x[8]]||0) + 1;
  });
  var daftar = Object.keys(kab).map(function(key){
    var v = kab[key];
    v.belum = v.c[BELUM]||0;
    v.persen = pct(v.n - v.belum, v.n);
    return v;
  }).sort(function(a,b){
    if(b.belum!==a.belum) return b.belum-a.belum;
    return a.nama<b.nama?-1:1;
  });
  $('wilKet').textContent = daftar.length + ' kab/kota';
  $('wilList').innerHTML = daftar.length ? daftar.map(function(x){
    var seg = URUT.map(function(k){
      var n = x.c[k]||0;
      return n ? '<i data-w="'+(n/x.n*100)+'" style="background:'+STAT[k].c+'"></i>' : '';
    }).join('');
    return '<div class="row"><div class="top"><span class="grow"><b>'+esc(x.nama)+'</b></span>' +
      '<b class="num">'+nf(x.belum)+'</b></div>' +
      '<div class="stack">'+seg+'</div>' +
      '<small>'+x.belum+' dari '+x.n+' sekolah belum kirim · sudah kirim '+satu(x.persen)+'%</small></div>';
  }).join('') : '<div class="kosong">Belum ada sekolah.</div>';
  /* legenda ditaruh di luar kotak bergulir supaya selalu terlihat */
  $('wilLeg').innerHTML = daftar.length ? URUT.map(function(k){
    return '<span><i class="dot" style="background:'+STAT[k].c+'"></i>'+STAT[k].t+'</span>';
  }).join('') : '';
  animasi($('wilList'));
  var tb = daftar.reduce(function(s,x){ return s+x.belum; },0);
  var tn = daftar.reduce(function(s,x){ return s+x.n; },0);
  $('wilNote').textContent = 'Angka di kanan tiap baris adalah jumlah sekolah yang belum mengirim ' +

    nf(tb) + ' dari ' + nf(tn) + ' sekolah belum kirim.';
}

/* Panel di bawah nama fasil: antrean reviunya dulu, karena itu yang bisa dia
   kerjakan sekarang, lalu sekolah K3 yang belum mengirim. */
function panel(x){
  var q = x.q.slice().sort(function(a,b){ return (b.u||0)-(a.u||0); });
  var belum = x.sek.filter(function(s){ return s[8]===BELUM; })
    .sort(function(a,b){ return (a[14]||1e9)-(b[14]||1e9); });
  var h = '<div class="panel">' +
    '<div class="panel-kepala"><b>'+q.length+' laporan menunggu reviu</b>' +
      '<span>'+x.lewat+' lewat '+AMBANG+' hari'+(x.mx!=null?' · terlama '+satu(x.mx)+' hari':'')+'</span>' +
      '<button type="button" class="det-tutup" data-tutup="1">Tutup</button></div>' +
    '<div class="panel-wa">'+tombolWa(x.hp,'WhatsApp '+x.nama,'wa-fasil') +
      (x.hp2 ? ' ' + tombolWa(x.hp2,'Nomor lain','wa-alt') : '') + '</div>';
  h += q.length ? q.map(function(a){
    return '<div class="s-row"><span class="grow"><b>'+esc(a.nm)+'</b>' +
      '<small>'+esc(a.np)+' · '+esc(a.kab)+(a.k3?'':' · di luar sasaran K3')+'</small>' +
      pil(PROSES, ' ' + satu(a.u) + ' hari') + '</span></div>';
  }).join('') : '<div class="kosong">Tidak ada laporan yang menunggu reviunya.</div>';
  if(belum.length){
    h += '<div class="panel-kepala" style="padding-top:14px"><b>'+belum.length+
      ' sekolah K3 belum kirim</b><span>perlu didorong</span></div>';
    h += belum.slice(0,25).map(function(s){
      return '<div class="s-row"><span class="grow"><b>'+esc(s[1])+'</b>' +
        '<small>'+esc(s[0])+' · '+esc(s[3])+'<br>Bendahara '+esc(s[13]||'-')+'</small>' +
        pilHadir(s) +
        '</span><span class="aksi">'+tombolWa(s[12],'WA kepsek')+'</span></div>';
    }).join('');
    if(belum.length>25) h += '<div class="kosong">25 teratas ditampilkan dari '+belum.length+
      ' sekolah. Sisanya ada di halaman Sekolah.</div>';
  }
  if(x.luarBelum) h += '<div class="kosong">Fasil ini juga membina '+x.luarBelum+
    ' sekolah di luar sasaran K3 yang belum mengirim Laporan 50%.</div>';
  return h + '</div>';
}
